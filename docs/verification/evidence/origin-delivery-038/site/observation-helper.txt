(() => {
  const h1 = document.querySelector('h1');
  const releasePage = location.pathname === '/whats-new';
  const articles = releasePage ? [...document.querySelectorAll('main#main article')].map(article => ({
    version: article.querySelector(':scope > p span')?.textContent?.trim() ?? null,
    title: article.querySelector('h2')?.textContent?.trim() ?? null,
    date: article.querySelector('time')?.getAttribute('datetime') ?? null,
    changes: [...article.querySelectorAll('li')].map(item => ({ title: item.querySelector('h3')?.textContent?.trim() ?? null, body: item.querySelector('p')?.textContent?.trim() ?? null })),
  })) : [];
  return {
    path: location.pathname,
    cssWidth: innerWidth,
    cssHeight: innerHeight,
    clientWidth: document.documentElement.clientWidth,
    scrollWidth: document.documentElement.scrollWidth,
    scrollHeight: document.documentElement.scrollHeight,
    fonts: document.fonts.status,
    reducedMotion: matchMedia('(prefers-reduced-motion: reduce)').matches,
    preferredDark: matchMedia('(prefers-color-scheme: dark)').matches,
    htmlClass: document.documentElement.className,
    h1: h1?.textContent?.trim() ?? null,
    headingColor: h1 ? getComputedStyle(h1).color : null,
    images: [...document.images].map(image => ({ path: new URL(image.currentSrc || image.src, location.href).pathname, loaded: image.complete && image.naturalWidth > 0, width: image.getBoundingClientRect().width, height: image.getBoundingClientRect().height })),
    alerts: [...document.querySelectorAll('[role="alert"], [role="status"]')].map(node => node.textContent?.trim() ?? ''),
    downloadLinks: [...new Set([...document.querySelectorAll('a[href]')].map(anchor => anchor.href).filter(href => href.startsWith('https://origin.imbored.fun/downloads/')))],
    articles,
    currentVersionDisplay: articles[0]?.version ?? null, currentVersion: articles[0]?.version?.replace(/^v/, '') ?? null,
    currentNotes: articles[0]?.changes.length ?? null,
    canonicalNotes: articles.reduce((total, article) => total + article.changes.length, 0),
  };
})()
