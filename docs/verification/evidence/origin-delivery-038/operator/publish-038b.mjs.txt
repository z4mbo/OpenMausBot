function command(file, args, name, input) {
  const startedAt = new Date().toISOString();
  const result = spawnSync(file, args, { cwd: root, input, maxBuffer: 4 * 1024 * 1024, timeout: 240000, windowsHide: true });
  if (name) {
    writeFileSync(resolve(root, `${folder}/${name}.log`), result.stdout ?? Buffer.alloc(0), { flag: 'wx' });
    writeFileSync(resolve(root, `${folder}/${name}.stderr.txt`), result.stderr ?? Buffer.alloc(0), { flag: 'wx' });
    writeFileSync(resolve(root, `${folder}/${name}-result.json`), JSON.stringify({ startedAt, finishedAt: new Date().toISOString(), exitCode: result.status, source, version: pins.version }, null, 2), { flag: 'wx' });
  }
  assert.equal(result.status, 0, `Operation failed: ${name ?? file}; inspect actual outcome before retry`);
  return result.stdout;
}
