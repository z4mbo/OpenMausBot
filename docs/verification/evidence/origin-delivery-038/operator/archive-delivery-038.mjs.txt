function archive(from, proposed, expected) {
  const absolute = resolve(from), originalPath = relative(root, absolute).replaceAll('\\', '/');
  if (originals.has(absolute)) return;
  assert.ok(lstatSync(absolute).isFile() && !lstatSync(absolute).isSymbolicLink());
  const bytes = readFileSync(absolute);
  assert.ok(bytes.length <= 16 * 1024 * 1024);
  if (expected) { assert.equal(bytes.length, expected.bytes); assert.equal(sha(bytes), expected.sha256); }
  let name = proposed.replaceAll('\\', '/');
  assert.ok(!name.startsWith('/') && !name.split('/').includes('..'));
  assert.ok(!/\.env(?:\.|$)/i.test(name));
  if (/\.(?:mjs|cjs|py|sh|ts)$/.test(name)) name += '.txt';
  const target = resolve(destination, name);
  assert.ok(!relative(destination, target).startsWith('..') && !existsSync(target));
  mkdirSync(dirname(target), { recursive: true });
  writeFileSync(target, bytes, { flag: 'wx' });
  assert.equal(sha(readFileSync(target)), sha(bytes));
  files.push({ path: name, originalPath, bytes: bytes.length, sha256: sha(bytes) });
  originals.set(absolute, name);
}

function walk(folder, prefix = '') {
  for (const entry of readdirSync(folder, { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))) {
    assert.ok(!entry.isSymbolicLink());
    const name = prefix + entry.name, absolute = resolve(folder, entry.name);
    if (entry.isDirectory()) walk(absolute, `${name}/`);
    else archive(absolute, destinationFor(name));
  }
}
