import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { checkDelivery038 } from './delivery-preparation-038b/finalize-delivery-038c.mjs';

const root = resolve(import.meta.dirname, '..');
const folder = '.omb-scratch/delivery-evidence-038';
const sha = bytes => createHash('sha256').update(bytes).digest('hex');
assert.equal(process.version, 'v24.21.0');
const input = JSON.parse(readFileSync(resolve(root, '.omb-scratch/delivery-preparation-038b/finalize-inputs-038b.template.json')));
input.templateOnly = false;
for (const descriptor of Object.values(input.proofs)) descriptor.sha256 = sha(readFileSync(resolve(root, descriptor.path)));
input.nativeAudit.path = `${folder}/native/receipt.json`;
const result = checkDelivery038(root, input);
assert.equal(result.receipt.status, 'published-and-proof-checked');
writeFileSync(resolve(root, `${folder}/finalize-inputs-038.json`), JSON.stringify(input, null, 2), { flag: 'wx' });
writeFileSync(resolve(root, `${folder}/delivery-receipt-038.json`), JSON.stringify(result.receipt, null, 2), { flag: 'wx' });
writeFileSync(resolve(root, `${folder}/final-audit-report-038.md`), result.report, { flag: 'wx' });
writeFileSync(resolve(root, `${folder}/finalizer-evidence-038.json`), JSON.stringify({ checkedAt: new Date().toISOString(), mutation: result.mutation, stateWritesByChecker: result.stateWrites, qualification: result.qualification, evidence: result.evidence }, null, 2), { flag: 'wx' });
console.log(JSON.stringify({ status: result.receipt.status, proofs: result.evidence.length, receiptSha256: sha(readFileSync(resolve(root, `${folder}/delivery-receipt-038.json`))) }));
