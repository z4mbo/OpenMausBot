import hashlib
import json
import re
import sys
from urllib.parse import quote, urlsplit
from urllib.request import Request, urlopen


def require(condition, message):
    if not condition:
        raise RuntimeError(message)


def unique_object(pairs):
    result = {}
    for key, value in pairs:
        require(key not in result, "Duplicate release metadata key")
        result[key] = value
    return result


def open_public(name, method="GET"):
    require(re.fullmatch(r"[A-Za-z0-9_.-]+", name), "Unsafe public filename")
    request = Request(
        "https://origin.imbored.fun/downloads/" + quote(name),
        method=method,
        headers={"Accept-Encoding": "identity", "Cache-Control": "no-cache"},
    )
    response = urlopen(request, timeout=60)
    target = urlsplit(response.url)
    require(response.status == 200, "Public HTTP response is not successful")
    require(target.scheme == "https" and target.hostname == "origin.imbored.fun" and target.port in (None, 443), "Unexpected public redirect")
    require(response.headers.get("Content-Encoding", "identity") == "identity", "Unexpected content encoding")
    return response


def metadata(name, expected_sha256):
    with open_public(name) as response:
        data = response.read(1024 * 1024 + 1)
    require(len(data) <= 1024 * 1024, "Oversized release metadata")
    require(hashlib.sha256(data).hexdigest() == expected_sha256, "Public metadata hash differs")
    return data


def stream_file(item):
    digest = hashlib.sha256()
    size = 0
    with open_public(item["name"]) as response:
        declared = response.headers.get("Content-Length")
        require(declared is None or int(declared) == item["size"], "Public Content-Length differs")
        while chunk := response.read(1024 * 1024):
            size += len(chunk)
            require(size <= item["size"], "Public payload exceeds the frozen size")
            digest.update(chunk)
    require(size == item["size"], "Public payload is truncated")
    require(digest.hexdigest() == item["sha256"], "Public payload hash differs")
    return {"name": item["name"], "size": size, "sha256": digest.hexdigest()}


def main(args):
    require(len(args) == 5, "Expected version, source, native run, manifest SHA-256 and checksums SHA-256")
    version, source, native_run, manifest_hash, checksums_hash = args
    require(re.fullmatch(r"\d+\.\d+\.\d+", version), "Unexpected release version")
    require(re.fullmatch(r"[0-9a-f]{40}", source), "Unexpected source identity")
    require(native_run.isdecimal(), "Unexpected native run identity")
    require(all(re.fullmatch(r"[0-9a-f]{64}", item) for item in [manifest_hash, checksums_hash]), "Unexpected metadata digest")
    manifest = json.loads(metadata(f"release-manifest-{version}.json", manifest_hash), object_pairs_hook=unique_object)
    require(manifest["version"] == version and manifest["tag"] == f"origin-v{version}", "Public version identity differs")
    require(manifest["commit"] == source and type(manifest["desktopRun"]) is int and manifest["desktopRun"] == int(native_run), "Public build identity differs")
    suffixes = ["arm64.dmg", "arm64.dmg.blockmap", "arm64.zip", "arm64.zip.blockmap", "setup.exe", "setup.exe.blockmap", "windows-x64.zip", "x64.dmg", "x64.dmg.blockmap", "x64.zip", "x64.zip.blockmap"]
    expected_names = {f"Origin-{version}-{suffix}" for suffix in suffixes} | {"latest.yml", "latest-mac.yml"}
    items = manifest["files"]
    require(len(items) == 13 and {item["name"] for item in items} == expected_names, "Public payload file set differs")
    checksums = metadata(f"SHA256SUMS-{version}.txt", checksums_hash).decode("utf-8")
    entries = []
    for line in checksums.splitlines():
        matched = re.fullmatch(r"([0-9a-f]{64})  ([A-Za-z0-9_.-]+)", line)
        require(matched is not None, "Malformed public checksum entry")
        entries.append((matched[2], matched[1]))
    require(len(entries) == 13 and dict(entries) == {item["name"]: item["sha256"] for item in items}, "Public checksum file differs")
    results = []
    for item in items:
        require(type(item["size"]) is int and 0 < item["size"] <= 1024 * 1024 * 1024, "Unexpected payload size")
        require(re.fullmatch(r"[0-9a-f]{64}", item["sha256"]), "Unexpected payload digest")
        result = stream_file(item)
        results.append(result)
        print(json.dumps({"verifiedPublicPayload": result}), flush=True)
    by_name = {item["name"]: item for item in items}
    aliases = {"Origin-Setup.exe": "setup.exe", "Origin-Mac-AppleSilicon.dmg": "arm64.dmg", "Origin-Mac-Intel.dmg": "x64.dmg"}
    for alias, suffix in aliases.items():
        result = stream_file({**by_name[f"Origin-{version}-{suffix}"], "name": alias})
        print(json.dumps({"verifiedPublicAlias": result}), flush=True)
    print(json.dumps({"status": "passed", "version": version, "source": source, "desktopRun": int(native_run), "payloads": len(results), "aliases": len(aliases), "metadata": 2, "payloadBytes": sum(item["size"] for item in results)}), flush=True)


if __name__ == "__main__":
    main(sys.argv[1:])
