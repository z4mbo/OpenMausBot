import assert from 'node:assert/strict';
import { execFileSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import { writeFileSync } from 'node:fs';
import { stripTypeScriptTypes } from 'node:module';
import vm from 'node:vm';
import { deriveFrozenExpectations } from './frozen-expectations-038.mjs';

const source = 'ab0cd72b4c2f004c1015adca034472241569c04b';
const expectations = deriveFrozenExpectations(source);
const code = stripTypeScriptTypes(execFileSync('git', ['show', `${source}:shared/origin-releases.ts`], { encoding: 'utf8' })).replace(/^\s*export\s+/gm, '');
assert.ok(!/^\s*import\b/m.test(code));
const history = JSON.parse(vm.runInNewContext(`${code}\nJSON.stringify(releases)`, {}, { timeout: 1000 }));
const textContent = value => value.replace(/<[^>]*>/g, '').replace(/&(amp|lt|gt|quot|#x27|#39);/g, (_, entity) => ({ amp: '&', lt: '<', gt: '>', quot: '"', '#x27': "'", '#39': "'" })[entity]).replace(/\s+/g, ' ').trim();
const routes = [];
for (const [path, heading] of [['/', 'Every project gets a team of agents that ships.'], ['/login', 'Sign in to Origin'], ['/signup', 'Create your Origin account'], ['/whats-new', "What's new"]]) {
  const response = await fetch(`https://origin.imbored.fun${path}`, { headers: { 'Cache-Control': 'no-cache', 'Accept': 'text/html' }, redirect: 'error', signal: AbortSignal.timeout(60_000) });
  assert.equal(response.status, 200, path);
  assert.ok(response.headers.get('content-type')?.includes('text/html'));
  const html = await response.text();
  assert.ok(html.length < 4 * 1024 * 1024);
  const h1 = /<h1\b[^>]*>([\s\S]*?)<\/h1>/.exec(html);
  const authShell = path === '/login' || path === '/signup';
  if (authShell) {
    assert.equal(h1, null, `Initial auth shell: ${path}`);
    assert.ok(html.includes('role="status"') && html.includes('Loading'));
  } else {
    assert.ok(h1, `Heading: ${path}`);
    assert.equal(textContent(h1[1]), heading, path);
  }
  assert.ok(!html.includes('This preview uses sample data.'));
  const result = { path, status: response.status, bytes: Buffer.byteLength(html), sha256: createHash('sha256').update(html).digest('hex'), heading: authShell ? null : heading, authShellOnly: authShell };
  if (path === '/') {
    for (const name of ['Origin-Setup.exe', 'Origin-Mac-AppleSilicon.dmg', 'Origin-Mac-Intel.dmg']) assert.ok(html.includes(`https://origin.imbored.fun/downloads/${name}`), name);
    assert.ok(html.includes('/landing/issues.webp'));
    result.downloadLinks = 3;
  }
  if (path === '/whats-new') {
    assert.equal((html.match(/<article(?:\s|>)/g) ?? []).length, expectations.releaseArticles);
    assert.equal((html.match(/<h3(?:\s|>)/g) ?? []).length, expectations.canonicalTitleTotal);
    assert.deepEqual([...html.matchAll(/<h3\b[^>]*>([\s\S]*?)<\/h3>/g)].map(match => textContent(match[1])), history.flatMap(release => release.changes.map(change => change.title)));
    assert.deepEqual([...html.matchAll(/<h3\b[^>]*>[\s\S]*?<\/h3>\s*<p\b[^>]*>([\s\S]*?)<\/p>/g)].map(match => textContent(match[1])), history.flatMap(release => release.changes.map(change => change.body.replace(/\s+/g, ' ').trim())));
    const versions = [...html.matchAll(/<article\b[^>]*>([\s\S]*?)<\/article>/g)].map(match => {
      const version = /<span\b[^>]*>([\s\S]*?)<\/span>/.exec(match[1]);
      assert.ok(version);
      return textContent(version[1]);
    });
    assert.deepEqual(versions, history.map(release => `v${release.version}`));
    Object.assign(result, { currentNotes: expectations.currentNoteCount, canonicalTitles: expectations.canonicalTitleTotal, canonicalBodies: expectations.canonicalTitleTotal, articles: expectations.releaseArticles });
  }
  routes.push(result);
}
const result = { source, version: expectations.version, origin: 'https://origin.imbored.fun', anonymousHttp: true, credentialsOrCookiesSupplied: false, submissions: 0, expectations, routes, passed: true, checkedAt: new Date().toISOString() };
writeFileSync('.omb-scratch/delivery-evidence-038/public-http-038.json', JSON.stringify(result, null, 2) + '\n', { flag: 'wx' });
console.log(JSON.stringify(result, null, 2));
