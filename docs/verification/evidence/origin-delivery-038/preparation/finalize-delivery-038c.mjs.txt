const HERE = import.meta.dirname;
const sha = bytes => createHash('sha256').update(bytes).digest('hex');
const readLocal = name => readFileSync(resolve(HERE, name));
assert.equal(sha(readLocal('delivery-receipt-038b.template.json')), 'bce39ff3558ce8de75b78989effdaaacd41f0ba90c95766a6f2b3335fa44ad95', 'Reviewed receipt template changed');
assert.equal(sha(readLocal('evidence-selection-038b.json')), '309d04599b9d48ad3da7a46f6b92287c22ba61bdb1b17391d0994595ac5ce54e', 'Reviewed evidence selection changed');
assert.equal(sha(readLocal('final-audit-report-038b.template.md')), '85960f08fcd67fced0050b95bd43ae5c6a0cd044e5b06d58aa597f9aac0fc165', 'Reviewed report template changed');
assert.equal(sha(readFileSync(resolve(HERE, '../prepublication-037b.mjs'))), 'ba70dccb6dab5dee6b371e1edbb8035984ddd4e07564056b413d46d5bd38bcd6', 'Reviewed PR policy guard changed');
const template = JSON.parse(readLocal('delivery-receipt-038b.template.json'));
const pins = template.pins;
const PROOF_NAMES = ['prepublication', 'ci', 'github', 'githubMetadata', 'downloadsLog', 'downloadsResult', 'http', 'runtime', 'publishGithubResult', 'promoteDownloadsResult', 'promoteSiteResult', 'publicLanding', 'publicWhatsNew', 'publicWhatsNewAX'];
const NATIVE_SHA = '4c17e1451fbc8a66545516e94acef675cffbea74dcd7bf71545b2fde0a9d4e27';
const REVIEW_SHA = 'b29e414563dc8bfb0a0cb716321d42533d55becf7097eda4e290d9898698a0a1';
const SOURCE = 'ab0cd72b4c2f004c1015adca034472241569c04b';

export function checkDelivery038(rootArg, input) {
  const root = resolve(rootArg);
  assert.equal(input.templateOnly, false, 'Actual receipt inputs are still pending');
  assert.equal(input.source, SOURCE); assert.equal(pins.source, SOURCE); assert.equal(pins.version, '0.2.14');
  assert.deepEqual(Object.keys(input.proofs).sort(), [...PROOF_NAMES].sort());
  const evidence = [];
  function checked(descriptor, name, allowEmpty = false) {
    assert.equal(typeof descriptor?.path, 'string'); assert.match(descriptor.path, /^(?:\.omb-scratch|docs\/verification)\/[A-Za-z0-9_./-]+$/);
    assert.ok(!descriptor.path.split('/').includes('..') && !/\.env(?:\.|$)/i.test(descriptor.path));
    assert.match(descriptor.sha256 ?? '', /^[a-f0-9]{64}$/);
    const file = resolve(root, descriptor.path); assert.ok(!relative(root, file).startsWith('..'));
    const stat = lstatSync(file); assert.ok(stat.isFile() && !stat.isSymbolicLink() && stat.size >= (allowEmpty ? 0 : 1) && stat.size <= 16 * 1024 * 1024);
    const bytes = readFileSync(file); assert.equal(sha(bytes), descriptor.sha256, `Proof bytes changed: ${name}`);
    evidence.push({ name, path: descriptor.path, bytes: bytes.length, sha256: descriptor.sha256 }); return bytes;
  }
  const bytes = Object.fromEntries(PROOF_NAMES.map(name => [name, checked(input.proofs[name], name)]));
  const records = Object.fromEntries(PROOF_NAMES.filter(name => !['downloadsLog', 'runtime', 'publicLanding', 'publicWhatsNew', 'publicWhatsNewAX'].includes(name)).map(name => [name, JSON.parse(bytes[name])]));
  const initialFailureBytes = checked(input.initialCiFailure, 'initialCiFailure');
  assert.equal(sha(initialFailureBytes), 'd2e194bb1c65ff24284b822f38baf62a1ea8d0251913bdf30b613d427512e002');
  checked(input.initialFailureDiagnostics, 'initialFailureDiagnostics');
  assert.equal(input.initialFailureDiagnostics.sha256, 'a384de9121678da1f8f78431428ea80a4aa1fb9d24d3b957bffe82f0199a3fcc');
  const nativeBytes = checked(input.nativeAudit, 'nativeAudit'); assert.equal(sha(nativeBytes), NATIVE_SHA);
  const native = JSON.parse(nativeBytes);
  const nativeDirectory = posix.dirname(input.nativeAudit.path);
  const nativeManifest = JSON.parse(checked({ path: `${nativeDirectory}/manifest.json`, sha256: 'cceb7f99e51086de7badfb2a51f919e1476b5d9a43ea24e4de5aa13b93991b84' }, 'nativeManifest'));
  assert.equal(nativeManifest.sourceCommit, SOURCE); assert.equal(nativeManifest.nativeRun, pins.nativeRun); assert.equal(nativeManifest.version, pins.version);
  assert.equal(nativeManifest.totalFiles, 76); assert.equal(nativeManifest.entries.length, 76); assert.equal(new Set(nativeManifest.entries.map(item => item.path)).size, 76);
  let nativeTextBytes = 0;
  for (const item of nativeManifest.entries) { const bytes = checked({ path: `${nativeDirectory}/${item.path}`, sha256: item.sha256 }, `native/${item.path}`, item.bytes === 0); assert.equal(bytes.length, item.bytes); nativeTextBytes += bytes.length; }
  assert.equal(nativeTextBytes, nativeManifest.totalBytes); assert.equal(nativeTextBytes, 3085458);
  const previous = JSON.parse(checked({ path: template.previousPublic035.path, sha256: template.previousPublic035.sha256 }, 'immutable035Receipt'));
  const manifestBytes = checked({ path: '.omb-scratch/distribution/0.2.14/release-manifest-0.2.14.json', sha256: pins.manifestSha256 }, 'distributionManifest');
  checked({ path: '.omb-scratch/distribution/0.2.14/SHA256SUMS-0.2.14.txt', sha256: pins.checksumsSha256 }, 'distributionChecksums');
  const manifest = JSON.parse(manifestBytes);
  assert.equal(manifest.commit, SOURCE); assert.equal(manifest.version, pins.version); assert.equal(manifest.desktopRun, pins.nativeRun); assert.equal(manifest.files.length, 13);
  const payloadBytes = manifest.files.reduce((sum, file) => { assert.ok(Number.isSafeInteger(file.size) && file.size > 0); return sum + file.size; }, 0);
  const pre = records.prepublication;
  assert.equal(pre.source, SOURCE); assert.equal(pre.version, pins.version); assert.equal(pre.prepublication, 'passed'); assert.equal(pre.mutation, false);
  assert.equal(pre.ciPolicy.runAttempt, pins.ciAttempt); assert.equal(pre.ciPolicy.initialFailure.rootCause, 'unproved'); assert.equal(pre.ciPolicy.initialFailure.sourceFixClaimed, false); assert.equal(pre.ciPolicy.jobExecutionProvenance.length, 20);
  assert.equal(pre.ciPolicy.jobExecutionProvenance.filter(job => job.classification === 'retained-success-execution').length, 17); assert.equal(pre.ciPolicy.jobExecutionProvenance.filter(job => job.classification === 'retained-skipped-record').length, 1); assert.equal(pre.ciPolicy.jobExecutionProvenance.filter(job => job.classification === 'retried-execution').length, 2);
  assert.equal(pre.ciPolicy.expectedJobs, 20); assert.equal(pre.ciPolicy.mainVitestMacRun, false); assert.equal(pre.ciPolicy.checkout.kind, 'pull_request_merge'); assert.equal(pre.ciPolicy.checkout.head, SOURCE);
  assert.equal(pre.ciPolicy.checkout.baseIsAncestorOfHead, true); assert.match(pre.ciPolicy.checkout.tree, /^[a-f0-9]{40}$/);
  assert.equal(pre.backendReused, true); assert.equal(pre.migrationsRequired, false); assert.equal(pre.providerReadinessRechecked, false);
  assert.deepEqual(pre.historical031, previous.backendDelivery.historical031); assert.deepEqual(pre.historicalMail, previous.mail);
  const ci = records.ci; assert.equal(ci.source, SOURCE);
  for (const [key, id, names] of [['ci', pins.ciRun, CI_JOBS037], ['native', pins.nativeRun, NATIVE_JOBS037], ['backendSite', pins.backendSiteRun, BACKEND_SITE_JOBS037]]) {
    const run = ci[key]; assert.equal(run.id, id); assert.equal(run.attempt, key === 'ci' ? pins.ciAttempt : key === 'native' ? pins.nativeAttempt : pins.backendSiteAttempt); assert.equal(run.source, SOURCE); assert.equal(run.status, 'completed'); assert.equal(run.conclusion, 'success');
    assert.deepEqual(run.jobs.map(job => job.name).sort(), [...names].sort());
    for (const job of run.jobs) { assert.equal(job.status, 'completed'); assert.equal(job.conclusion, job.name === 'deploy composio broker (Cloudflare Worker)' ? 'skipped' : 'success'); }
  }
  assert.equal(ci.ci.jobs.filter(job => job.name.startsWith('vitest (')).length, 8);
  for (const job of pre.ciPolicy.jobExecutionProvenance) { assert.equal(job.apiRunAttempt, pins.ciAttempt); assert.equal(ci.ci.jobs.find(item => item.name === job.name).id, job.latestJobId); }
  const gh = records.github, metadata = records.githubMetadata;
  for (const [key, value] of Object.entries({ id: pins.githubReleaseId, version: pins.version, source: SOURCE, nativeRun: pins.nativeRun, draft: false, assets: 15, payloads: 13, sha256Verified: true })) assert.equal(gh[key], value);
  assert.equal(metadata.id, pins.githubReleaseId); assert.equal(metadata.draft, false); assert.equal(metadata.prerelease, false); assert.equal(metadata.tag_name, 'origin-v0.2.14');
  assert.equal(metadata.html_url, 'https://github.com/z4mbo/Origin/releases/tag/origin-v0.2.14');
  assert.ok(Number.isFinite(Date.parse(metadata.published_at))); assert.equal(sha(Buffer.from(metadata.body)), '6834433ea45911633fd0abcd018e0b372932baad11a4c9b9375458f5a2e92cb8');
  const downloads = bytes.downloadsLog.toString().trim().split(/\r?\n/).map(JSON.parse), summary = downloads.at(-1);
  assert.deepEqual(summary, { status: 'passed', version: pins.version, source: SOURCE, desktopRun: pins.nativeRun, payloads: 13, aliases: 3, metadata: 2, payloadBytes });
  assert.equal(records.downloadsResult.exitCode, 0); assert.ok(Number.isFinite(Date.parse(records.downloadsResult.finishedAt)));
  const payloads = downloads.filter(line => line.verifiedPublicPayload).map(line => line.verifiedPublicPayload);
  assert.deepEqual(payloads.map(file => file.name).sort(), manifest.files.map(file => file.name).sort());
  for (const file of payloads) assert.deepEqual(file, manifest.files.find(item => item.name === file.name));
  const aliases = downloads.filter(line => line.verifiedPublicAlias).map(line => line.verifiedPublicAlias);
  const aliasTargets = { 'Origin-Setup.exe': 'setup.exe', 'Origin-Mac-AppleSilicon.dmg': 'arm64.dmg', 'Origin-Mac-Intel.dmg': 'x64.dmg' };
  assert.deepEqual(aliases.map(file => file.name).sort(), Object.keys(aliasTargets).sort());
  for (const alias of aliases) { const target = manifest.files.find(file => file.name === `Origin-0.2.14-${aliasTargets[alias.name]}`); assert.equal(alias.size, target.size); assert.equal(alias.sha256, target.sha256); }
  assert.equal(downloads.length, 17);
  const http = records.http; assert.equal(http.source, SOURCE); assert.equal(http.version, pins.version); assert.equal(http.passed, true);
  assert.equal(http.origin, 'https://origin.imbored.fun'); assert.equal(http.anonymousHttp, true); assert.equal(http.credentialsOrCookiesSupplied, false); assert.equal(http.submissions, 0);
  assert.deepEqual(http.routes.map(route => route.path).sort(), ['/', '/login', '/signup', '/whats-new'].sort());
  for (const route of http.routes) assert.equal(route.status, 200);
  const whatsNew = http.routes.find(route => route.path === '/whats-new');
  for (const [key, value] of Object.entries({ currentNotes: 4, canonicalTitles: 55, canonicalBodies: 55, articles: 7 })) assert.equal(whatsNew[key], value);
  assert.ok(bytes.publicWhatsNewAX.toString().includes('0.2.14'));
  for (const name of ['publicLanding', 'publicWhatsNew']) assert.ok(bytes[name].length > 10_000, 'Public screenshot missing visible evidence');
  for (const name of ['publishGithubResult', 'promoteDownloadsResult', 'promoteSiteResult']) { assert.equal(records[name].exitCode, 0); assert.ok(Number.isFinite(Date.parse(records[name].finishedAt))); }
  assert.ok(Date.parse(http.checkedAt) >= Date.parse(records.promoteSiteResult.finishedAt));
  assert.ok(Date.parse(records.downloadsResult.finishedAt) >= Date.parse(records.promoteDownloadsResult.finishedAt));
  const runtime = bytes.runtime.toString().trim().split(/\s+/); assert.equal(runtime.length, 6);
  assert.deepEqual(runtime.slice(0, 5), [pins.siteImageId, 'true', '0', SOURCE, pins.version]); assert.ok(Number.isFinite(Date.parse(runtime[5])));
  assert.equal(native.sourceCommit, SOURCE); assert.equal(native.version, pins.version); assert.equal(native.nativeRun, pins.nativeRun); assert.equal(native.nativeAttempt, 1); assert.equal(native.status, 'bounded-package-audit-passed');
  assert.equal(native.archiveDownloads.length, 2); for (const archive of native.archiveDownloads) { assert.equal(archive.apiAndActualMatch, true); assert.equal(archive.actualBytes, archive.apiReportedBytes); assert.equal('sha256:' + archive.actualSha256, archive.apiReportedDigest); }
  assert.equal(native.distribution.manifest.sha256, pins.manifestSha256); assert.equal(native.distribution.checksums.sha256, pins.checksumsSha256); assert.equal(native.distribution.payloadBytes, payloadBytes);
  assert.equal(native.packageAudit.plainElectronComparisons, 477); assert.equal(native.packageAudit.targets.length, 3); assert.equal(native.packageAudit.bundledOwnership.targets, 3); assert.equal(native.packageAudit.bundledOwnership.sourceBodiesEquivalent, true); assert.equal(native.packageAudit.bundledOwnership.requiredRouteAndLifecycleCallSitesVerified, true);
  assert.equal(native.windowsResourceSmoke.exitCode, 0); assert.equal(native.windowsResourceSmoke.ownedProcessesClosed, true); assert.equal(native.windowsResourceSmoke.healthPortClosed, true); assert.equal(native.windowsResourceSmoke.executableGuiLaunched, false);
  const selection = JSON.parse(readLocal('evidence-selection-038b.json'));
  for (const item of selection.files) {
    if (item.root === 'readOnlyFinalCheckout') checked(item, item.proposedPermanentPath);
    else { assert.equal(item.root, 'privatePreparationWorktree'); const bytes = readFileSync(item.path); assert.equal(bytes.length, item.bytes); assert.equal(sha(bytes), item.sha256); }
  }
  const reviewItem = selection.files.find(file => file.path.endsWith('site-visual-review-038.json'));
  assert.equal(reviewItem.sha256, REVIEW_SHA); const reviewBytes = readFileSync(reviewItem.path); assert.equal(sha(reviewBytes), REVIEW_SHA);
  const review = JSON.parse(reviewBytes); assert.equal(review.source, SOURCE); assert.equal(review.siteImageId, pins.siteImageId); assert.equal(review.acceptedVisualStates, 20); assert.equal(review.rejectedCaptures, 0); assert.deepEqual(review.pendingEvidence, []);
  const receipt = structuredClone(template); delete receipt.templateOnly;
  Object.assign(receipt, { status: 'published-and-proof-checked', generatedAt: new Date().toISOString(), published: true, publishedAt: metadata.published_at, siteDeployed: true, artifactFeedsVerified: true, ciConclusion: 'success', nativeConclusion: 'success', prepublicationPassedAt: pre.checkedAt });
  receipt.ciRetry.status = 'TARGETED_RETRY_ACCEPTED_WITH_INITIAL_CAUSE_UNPROVED'; receipt.ciRetry.initialFailure.evidence = input.initialCiFailure; receipt.ciRetry.initialFailure.diagnostics = input.initialFailureDiagnostics; receipt.ciRetry.targetedRetry.succeeded = true; receipt.ciRetry.targetedRetry.actualAttempt = pins.ciAttempt; receipt.ciRetry.targetedRetry.jobProvenance = structuredClone(pre.ciPolicy.jobExecutionProvenance);
  receipt.ciTerminalEvidence = input.proofs.ci; receipt.nativeTerminalEvidence = input.proofs.ci; receipt.backendSiteEvidence = input.proofs.ci;
  receipt.ciMainShardResults = { platforms: ['ubuntu-latest', 'windows-latest'], shards: 8, macMainVitestRun: false, results: structuredClone(ci.ci.tests ?? null), qualification: 'Original CI summary/count qualification retained; platform occurrences are not unique tests. Latest accepted shards combine retained first-attempt executions and the targeted retry; no new full matrix is claimed.' };
  receipt.github = { ...gh, publishedAt: metadata.published_at, bodyVerified: true, evidence: input.proofs.github };
  receipt.publicDownloads = { ...summary, verifiedAt: records.downloadsResult.finishedAt, evidence: input.proofs.downloadsLog };
  receipt.sitePromotion = { runningImage: runtime[0], runningSource: runtime[3], previousImage: pins.previousSiteImageId, promotedAt: records.promoteSiteResult.finishedAt, containerStartedAt: runtime[5], restartCount: 0 };
  receipt.publicSite = { ...http, accountSubmissions: 0, browserRoutes: ['/', '/whats-new'], initialAuthHttpShellOnly: true, evidence: input.proofs.http };
  receipt.nativeAudit = { receiptSha256: NATIVE_SHA, source: SOURCE, nativeRun: pins.nativeRun, status: native.status, plainElectronComparisons: 477, bundledOwnershipTargets: 3, actualArchivesHashed: true, windowsResourceSmokeExit: 0, installedWindowsGui: false, physicalMacExecution: false, limits: native.limits };
  receipt.backendDelivery.mode = 'reused-existing-deployed-backend'; receipt.backendDelivery.finalFrozenClosureVerified = true; receipt.backendDelivery.closureEvidence = input.proofs.prepublication;
  let report = `# Origin 0.2.14 — verified delivery\n\nOrigin 0.2.14 was published from **${SOURCE}** at **${metadata.published_at}**. The matching website promotion completed at **${records.promoteSiteResult.finishedAt}**, running **${pins.siteImageId}**. Public HTTPS byte checks verified 13 payloads, two metadata files and three aliases; GitHub has the matching 15 assets. Manifest SHA256: ${pins.manifestSha256}; checksums SHA256: ${pins.checksumsSha256}.\n\nCanonical PR CI ${pins.ciRun} passed all 20 expected jobs (19 success, intended broker skip), including eight Ubuntu/Windows main shards and required Mac smokes. Main Mac Vitest was not run under PR policy. The tested PR merge checkout **${pre.ciPolicy.checkout.checkout}** has the same tree as frozen source **${SOURCE}**. Native ${pins.nativeRun} and backend/site tag ${pins.backendSiteRun} passed at that source; no duplicate manual matrix is implied.\n\nActual native archives were independently hashed. The package audit passed 477 Electron source comparisons and three emitted ownership-definition/call-site checks, static Mac architecture/metadata/feed inspection and an isolated Windows Node-resource smoke. Installed GUI/updater and physical Mac execution remain unverified.\n\nFresh compiled-site acceptance covers 20 reviewed states at 375/1440 in both themes, including initial signup policy loading and settled retry errors. Public browser evidence covers landing and What's new; anonymous HTTP checks cover four routes and all 55 canonical titles/bodies across seven articles. No account was submitted. No 035 RootFS reuse is claimed.\n\n${readLocal('final-audit-report-038b.template.md').toString().split('## Verified scope prepared for delivery')[1].split('## Delivery pending')[0].replace('Terminal CI and publication evidence are still required before delivery acceptance.', 'Terminal CI, package and publication evidence are bound in this receipt.')}\n`;
  report = report.replace('Canonical PR CI', `Attempt 1 failed in Windows shard 4: the provider-key-flow SSE predicate did not match within its 10-second deadline. The cause remains unproved, and no source fix is claimed. The targeted retry used attempt ${pins.ciAttempt} for that shard and the dependent aggregate. GitHub assigned attempt-${pins.ciAttempt} IDs to copied job records too: 17 successful executions and one skipped broker record are retained from attempt 1, while two jobs actually reran. Original failure/diagnostic evidence is preserved. Source, native artifacts and tag did not change.\n\nCanonical PR CI`);
  return { mutation: false, stateWrites: 0, receipt, report, evidence, qualification: 'Check-only validation of explicit hashed evidence files. This helper makes no network calls and writes no files. Publication and production actions remain attributable to the root records.' };
}
