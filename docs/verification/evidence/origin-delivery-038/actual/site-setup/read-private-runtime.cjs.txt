const assert = require('node:assert/strict');
const fs = require('node:fs');
const path = require('node:path');
const cp = require('node:child_process');
const frozen = 'ab0cd72b4c2f004c1015adca034472241569c04b';
const image = 'sha256:b0a0a3315c8eeec4ea8e337b935142cf04142962e3cb92dcde87369f74684201';
const name = 'origin-site-038-preview';
const inspect = (kind, target, template) => JSON.parse(cp.execFileSync('ssh.exe', ['-o', 'BatchMode=yes', '-o', 'ConnectTimeout=10', 'hetzner', `docker ${kind} inspect --format '${template}' ${target}`], { encoding: 'utf8' }));
const container = inspect('container', name, '{"id":{{json .Id}},"image":{{json .Image}},"status":{{json .State.Status}},"restarts":{{json .RestartCount}},"labels":{{json .Config.Labels}},"mounts":{{json .Mounts}},"networks":{{json .NetworkSettings.Networks}},"ports":{{json .HostConfig.PortBindings}},"capDrop":{{json .HostConfig.CapDrop}},"securityOpt":{{json .HostConfig.SecurityOpt}},"privileged":{{json .HostConfig.Privileged}}}');
const network = inspect('network', name, '{"id":{{json .Id}},"name":{{json .Name}},"internal":{{json .Internal}},"labels":{{json .Labels}}}');
const built = inspect('image', image, '{"id":{{json .Id}},"labels":{{json .Config.Labels}},"rootFS":{{json .RootFS}},"workingDir":{{json .Config.WorkingDir}},"command":{{json .Config.Cmd}},"exposedPorts":{{json .Config.ExposedPorts}}}');
assert.equal(container.image, image); assert.equal(built.id, image); assert.equal(container.status, 'running');
assert.equal(container.labels['origin.preview.source'], frozen); assert.equal(container.labels['origin.preview.version'], '0.2.14');
assert.equal(built.labels['org.opencontainers.image.revision'], frozen); assert.equal(built.labels['org.opencontainers.image.version'], '0.2.14');
assert.deepEqual(container.mounts, []); assert.equal(container.privileged, false); assert.deepEqual(container.capDrop, ['ALL']);
assert(container.securityOpt.includes('no-new-privileges')); assert.equal(network.internal, true);
assert.deepEqual(container.ports, { '3000/tcp': [{ HostIp: '127.0.0.1', HostPort: '33038' }] });
assert.deepEqual(Object.keys(container.networks), [name]);
const privateIP = container.networks[name].IPAddress; assert.match(privateIP, /^172\.\d{1,3}\.\d{1,3}\.\d{1,3}$/);
const proof = { source: frozen, image, version: '0.2.14', container, network, built, privateIP, capturedAt: new Date().toISOString(), scope: 'Only the uniquely owned038 private image/container/network were inspected; no Env values were retrieved.' };
fs.writeFileSync(path.join(__dirname, 'runtime.json'), JSON.stringify(proof, null, 2) + '\n', { flag: 'wx' });
console.log(JSON.stringify({ source: frozen, image, containerId: container.id, privateIP, internal: network.internal, mounts: container.mounts.length, loopbackOnly: true }));
