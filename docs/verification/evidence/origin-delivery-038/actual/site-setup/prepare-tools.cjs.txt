const fs = require('node:fs');
const path = require('node:path');
const cp = require('node:child_process');
const crypto = require('node:crypto');
const own = path.resolve(__dirname, '../..');
const sourceRoot = 'C:\\Users\\aless\\Documents\\Origin-bugfix';
const frozen = 'ab0cd72b4c2f004c1015adca034472241569c04b';
const sha = data => crypto.createHash('sha256').update(data).digest('hex');
if (cp.execFileSync('git', ['rev-parse', 'HEAD'], { cwd: own }).toString().trim() !== frozen) throw new Error('Not the final frozen checkout');
if (cp.execFileSync('git', ['status', '--porcelain=v1'], { cwd: own }).length) throw new Error('Checkout must be clean');
const names = ['prepare-site-034.mjs', 'frozen-expectations-034.mjs', 'build-site-034.sh', 'preview-site-034.sh', 'serve-site-034-preview.mjs', 'prepare-site-identity.mjs', 'release-contract.mjs'];
const adapted = [];
for (const name of names) {
  const input = path.join(sourceRoot, '.omb-scratch', name);
  const before = fs.readFileSync(input);
  const copyName = name.replace('034', '038');
  let contents = before.toString('utf8');
  if (name.includes('034')) contents = contents.replaceAll('0.2.13', '0.2.14').replaceAll('-034', '-038').replaceAll('candidate034', 'candidate038').replaceAll('33034', '33038')
    .replaceAll('/root/origin-release-staging/0.2.14', '/root/origin-release-staging/0.2.14-cycle038')
    .replaceAll('origin-audit-site:0.2.14-final', 'origin-audit-site:0.2.14-038-final')
    .replaceAll('site-release-identity-0.2.14.json', 'site-release-identity-0.2.14-038.json');
  const output = path.join(own, '.omb-scratch', copyName);
  fs.writeFileSync(output, contents, { flag: 'wx' });
  if (!fs.readFileSync(input).equals(before)) throw new Error(`Original changed during read: ${name}`);
  adapted.push({ original: input, originalSha256: sha(before), output: path.relative(own, output).replaceAll('\\', '/'), adaptedSha256: sha(fs.readFileSync(output)) });
}
fs.writeFileSync(path.join(__dirname, 'helpers.json'), JSON.stringify({ source: frozen, version: '0.2.14', cycle: '038', adapted, originalFilesUnchanged: true, productionActions: 0 }, null, 2) + '\n', { flag: 'wx' });
console.log(JSON.stringify({ source: frozen, helpers: adapted.length, originalFilesUnchanged: true, productionActions: 0 }));
