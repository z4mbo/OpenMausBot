function canonHash(node) {

  const program = ts.createProgram([filename], options, host), checker = program.getTypeChecker();

  const symbols = new Map();
  const symbolId = identifier => {
    const symbol = checker.getSymbolAtLocation(identifier);
    const local = symbol?.declarations?.some(declaration => declaration.pos >= root.pos && declaration.end <= root.end);
    if (!local) return ['Global', expectedName(identifier.text)];
    if (!symbols.has(symbol)) symbols.set(symbol, symbols.size);
    return ['Local', symbols.get(symbol)];
  };
  const key = name => ts.isIdentifier(name) || ts.isStringLiteralLike(name) ? ['Property', name.text] : canonical(name);
  function canonical(item) {
    if (ts.isParenthesizedExpression(item)) return canonical(item.expression);
    if (ts.isShorthandPropertyAssignment(item)) { const symbol = checker.getShorthandAssignmentValueSymbol(item); if (!symbol) throw new Error('Unbound shorthand'); if (!symbols.has(symbol)) symbols.set(symbol, symbols.size); return ['PropertyAssignment', key(item.name), ['Local', symbols.get(symbol)], item.objectAssignmentInitializer ? canonical(item.objectAssignmentInitializer) : null]; }
    if (ts.isPropertyAssignment(item)) return ['PropertyAssignment', key(item.name), canonical(item.initializer), null];
    if (ts.isBindingElement(item)) return ['BindingElement', item.dotDotDotToken ? true : false, key(item.propertyName ?? item.name), canonical(item.name), item.initializer ? canonical(item.initializer) : null];
    if (ts.isPropertyAccessExpression(item)) return ['PropertyAccessExpression', canonical(item.expression), key(item.name), Boolean(item.questionDotToken)];
    if (ts.isIdentifier(item)) return symbolId(item);
    if (ts.isStringLiteralLike(item)) return ['String', item.text];
    if (ts.isNumericLiteral(item)) return ['Number', Number(item.text.replaceAll('_', ''))];
    const children = []; ts.forEachChild(item, child => { children.push(canonical(child)); });
    return [ts.SyntaxKind[item.kind], ...(ts.isVariableDeclarationList(item) ? [item.flags & (ts.NodeFlags.Const | ts.NodeFlags.Let)] : []), children];
  }
  const tree = canonical(root), digest = sha(JSON.stringify(tree)); fs.writeFileSync(path.join(__dirname, 'canonical-' + digest + '.json'), JSON.stringify(tree, null, 2) + '\n'); return digest;
}
function declarations(ast) {
  const functions = new Map(), variables = new Map();

    if (ts.isFunctionDeclaration(node) && node.name) functions.set(node.name.text, node);
    if (ts.isVariableStatement(node)) for (const item of node.declarationList.declarations) if (ts.isIdentifier(item.name)) variables.set(item.name.text, item);
  }
  return { functions, variables };
}

}
