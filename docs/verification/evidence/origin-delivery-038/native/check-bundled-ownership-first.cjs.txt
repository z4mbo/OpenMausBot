function canonical(node) {
  if (ts.isParenthesizedExpression(node)) return canonical(node.expression);
  if (ts.isIdentifier(node)) return ['Identifier', expectedName(node.text)];
  if (ts.isStringLiteralLike(node)) return ['String', node.text];
  if (ts.isNumericLiteral(node)) return ['Number', Number(node.text.replaceAll('_', ''))];
  const children = []; ts.forEachChild(node, child => { children.push(canonical(child)); });
  return [ts.SyntaxKind[node.kind], ...(ts.isVariableDeclarationList(node) ? [node.flags & (ts.NodeFlags.Const | ts.NodeFlags.Let)] : []), children];
}

function declarations(ast) {
  const functions = new Map(), variables = new Map();
  for (const node of ast.statements) {
    if (ts.isFunctionDeclaration(node) && node.name) functions.set(node.name.text, node);
    if (ts.isVariableStatement(node)) for (const item of node.declarationList.declarations) if (ts.isIdentifier(item.name)) variables.set(item.name.text, item);
  }
  return { functions, variables };
}

function constantValue(node) {
  if (ts.isParenthesizedExpression(node)) return constantValue(node.expression);
  if (ts.isNumericLiteral(node)) return Number(node.text.replaceAll('_', ''));
  if (ts.isBinaryExpression(node) && node.operatorToken.kind === ts.SyntaxKind.AsteriskToken) return constantValue(node.left) * constantValue(node.right);
  throw new Error('Unsupported static lease constant expression');
}
