const fs=require('node:fs');
const path=require('node:path');
const crypto=require('node:crypto');
const expected=['Origin','OpenMausBot','openmausbot','OpenGrokBot','opengrokbot'];
const sha=bytes=>crypto.createHash('sha256').update(bytes).digest('hex');
const targets=[
  {label:'mac-arm64',file:'mac-arm64/Origin.app/Contents/Resources/server/index.js'},
  {label:'mac-x64',file:'mac-x64/Origin.app/Contents/Resources/server/index.js'},
  {label:'windows-x64',file:'windows/resources/server/index.js'},
];
const results=[];
for(const target of targets){
  const filename=path.join(__dirname,target.file);
  if(!fs.existsSync(filename)){results.push({...target,available:false,qualification:'Known index bundle not available in selected extraction; no fallback conclusion.'});continue;}
  const bytes=fs.readFileSync(filename),source=bytes.toString('utf8');
  const found=/function (cuaDescriptorPaths\d*)\([^)]*\)\s*\{([\s\S]{0,1500}?)\r?\n\}/.exec(source);
  if(!found){results.push({...target,bundleBytes:bytes.length,bundleSha256:sha(bytes),available:false,qualification:'Named helper not found within the bounded emitted-function pattern; no fallback conclusion or broader scan.'});continue;}
  const array=/\[\s*"Origin"\s*,[^\]]{0,250}\]/.exec(found[0]);
  if(!array)throw new Error(target.label+': helper present but fallback array not found');
  const actual=JSON.parse(array[0]);
  if(JSON.stringify(actual)!==JSON.stringify(expected))throw new Error(target.label+': actual bundled fallback order differs');
  results.push({...target,available:true,bundleBytes:bytes.length,bundleSha256:sha(bytes),functionName:found[1],functionLine:source.slice(0,found.index).split('\n').length,functionBytes:Buffer.byteLength(found[0]),functionSha256:sha(found[0]),expectedOrder:expected,actualOrder:actual,exactOrder:true,emittedFunction:found[0],qualification:'Static actual downloaded JS inspection only; no native platform execution or descriptor discovery acceptance inferred.'});
}
const report={frozenSource:'ab0cd72b4c2f004c1015adca034472241569c04b',nativeRun:39182331813,scope:'Three known packaged server/index.js entries only; Mac index.js is the sole additional selected JS entry per ZIP.',targets:results};
fs.writeFileSync(path.join(__dirname,'bundled-fallback-results.json'),JSON.stringify(report,null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify({available:results.filter(target=>target.available).length,targets:results.map(({label,available,exactOrder,functionLine})=>({label,available,exactOrder,functionLine}))},null,2));
