import { execFileSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync } from 'node:fs';
import { createRequire } from 'node:module';
import { dirname, join } from 'node:path';
import { normalizePath } from 'vite';
import { launchVerificationServer } from '../../scripts/control-omb.ts';
import { mountPreview, parkUntilSignal, REPO_ROOT } from '../../scripts/testing/preview-fixture.ts';
const before = process.argv.includes('--before');
if (process.argv.slice(2).some(arg => arg !== '--before')) throw new Error('Only --before is supported');
const relative = 'site/components/brand.tsx';
const baseline = 'ec28ec06e935b10a176f26586348e81831475e80';
const source = before ? execFileSync('git', ['show', `${baseline}:${relative}`], {cwd:REPO_ROOT}) : readFileSync(join(REPO_ROOT,relative));
const sha = bytes => createHash('sha256').update(bytes).digest('hex');
const fixture = await launchVerificationServer();
let preview;
try {
  const siteRequire = createRequire(join(REPO_ROOT,'site/package.json'));
  const adapter = normalizePath(join(REPO_ROOT,'.omb-scratch/session-scope-036/transport.tsx'));
  const target = normalizePath(join(REPO_ROOT,relative));
  const proof = {version:before?'before':'after',baseline,source:relative,sha256:sha(source),realAuthPage:true,realSessionHook:true,syntheticConvex:true,nativeBackend:false};
  preview = await mountPreview(fixture, {
    entry:'/.omb-scratch/session-scope-036/preview.tsx',route:'/__origin-session-scope.html',title:`Origin session scope — ${proof.version}`,hmr:false,
    cacheDir:join(fixture.info.dataDir,'session-scope-vite-cache'),
    plugins:[{name:'exact-session-source',enforce:'pre',load(id){if(normalizePath(id.split('?')[0])===target) return source.toString();}}],
    aliases:[
      {find:/^react(?=\/|$)/,replacement:dirname(siteRequire.resolve('react/package.json'))},
      {find:/^react-dom(?=\/|$)/,replacement:dirname(siteRequire.resolve('react-dom/package.json'))},
      {find:/^(?:convex\/react|next\/link|@\/lib\/convex-api|\.\/origin-site)$/,replacement:adapter},
      {find:/^@\//,replacement:normalizePath(join(REPO_ROOT,'site'))+'/'},
    ],
    extraRoutes:[{path:'/__session-scope-proof.json',handler(_req,res){res.setHeader('content-type','application/json');res.end(JSON.stringify(proof));}}],
  });
  const ready = {...fixture.info,...proof,previewUrl:preview.previewUrl};
  writeFileSync(join(REPO_ROOT,'.omb-scratch/session-scope-036',before?'browser-before-fixture.json':'browser-after-fixture.json'),JSON.stringify(ready,null,2)+'\n');
  console.log(JSON.stringify(ready));
  await parkUntilSignal();
} finally {await preview?.close();await fixture.close();}
