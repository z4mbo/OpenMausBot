const fs = require('node:fs');
const path = require('node:path');
const cp = require('node:child_process');
const crypto = require('node:crypto');
const root = path.resolve(__dirname, '../..');
const parent = 'C:\\Users\\aless\\Documents\\Origin-bugfix';
const digest = data => crypto.createHash('sha256').update(data).digest('hex');
const before = JSON.parse(fs.readFileSync(path.join(__dirname, 'bindings-before.json'), 'utf8'));
const sources = before.sources.map(item => ({ ...item, checkoutUnchanged: digest(fs.readFileSync(path.join(root, item.file))) === item.checkoutSha256 }));
const leases = before.leases.map(item => ({ ...item, unchanged: digest(fs.readFileSync(path.join(root, item.file))) === item.sha256 }));
const tests = before.tests.map(item => {
  const own = fs.readFileSync(path.join(root, item.file));
  const parentNow = fs.readFileSync(path.join(parent, item.file));
  const snapshot = path.basename(item.file) + '.txt';
  fs.writeFileSync(path.join(__dirname, snapshot), own);
  return { ...item, ownUnchanged: digest(own) === item.sha256, parentStillByteExact: own.equals(parentNow), snapshot };
});
const results = ['gate-red', 'connect-red'].map(label => {
  const data = JSON.parse(fs.readFileSync(path.join(__dirname, `${label}.json`), 'utf8'));
  const exit = JSON.parse(fs.readFileSync(path.join(__dirname, `${label}-exit.json`), 'utf8'));
  const cases = data.testResults.flatMap(suite => suite.assertionResults.map(test => ({ name: test.fullName, status: test.status, firstFailure: test.failureMessages[0]?.split('\n')[0] ?? null })));
  return { label, total: data.numTotalTests, passed: data.numPassedTests, failed: data.numFailedTests, pending: data.numPendingTests, runtimeErrors: data.numRuntimeErrorTestSuites ?? null, cases, exitCode: exit.code };
});
const proof = { ...before, sources, tests, leases, results, ownHeadAfter: cp.execFileSync('git', ['rev-parse', 'HEAD'], { cwd: root }).toString().trim(), trackedDiffAfter: cp.execFileSync('git', ['diff', '--name-only', 'HEAD'], { cwd: root }).toString().trim(), completedAt: new Date().toISOString() };
if (sources.some(item => !item.checkoutUnchanged) || leases.some(item => !item.unchanged) || tests.some(item => !item.ownUnchanged || !item.parentStillByteExact) || proof.ownHeadAfter !== before.ownHead || proof.trackedDiffAfter) throw new Error('source/test/lease binding changed during replay');
if (results.some(item => item.exitCode !== 1 || item.failed === 0 || item.runtimeErrors)) throw new Error('not a genuine behavior red replay');
fs.writeFileSync(path.join(__dirname, 'final-proof.json'), JSON.stringify(proof, null, 2) + '\n');
const names = [
  'REPORT.md', 'bindings-before.json', 'final-proof.json', 'prepare-replay.cjs', 'run-tests.cjs', 'finish-proof.cjs',
  ...tests.map(item => item.snapshot),
  ...['gate-red', 'connect-red'].flatMap(label => [`${label}.log`, `${label}.json`, `${label}-command.json`, `${label}-exit.json`]),
].sort();
const files = names.map(file => { const data = fs.readFileSync(path.join(__dirname, file)); return { file, size: data.length, sha256: digest(data) }; });
const manifest = Buffer.from(JSON.stringify({ schemaVersion: 1, files, count: files.length, totalBytes: files.reduce((total, item) => total + item.size, 0) }, null, 2) + '\n');
fs.writeFileSync(path.join(__dirname, 'manifest.json'), manifest);
for (const entry of files) { const data = fs.readFileSync(path.join(__dirname, entry.file)); if (data.length !== entry.size || digest(data) !== entry.sha256) throw new Error(`manifest mismatch: ${entry.file}`); }
console.log(JSON.stringify({ results: results.map(({ cases, ...item }) => item), files: files.length, bytes: files.reduce((total, item) => total + item.size, 0), manifestSha256: digest(manifest), leaseFilesUnchanged: true, parentTestsStillByteExact: true }, null, 2));
