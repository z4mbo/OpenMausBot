const fs = require('node:fs');
const path = require('node:path');
const cp = require('node:child_process');
const crypto = require('node:crypto');
const root = path.resolve(__dirname, '../..');
const sha256 = data => crypto.createHash('sha256').update(data).digest('hex');
const git = (...args) => cp.execFileSync('git', args, { cwd: root });
const source = ['server/origin-run-lease.ts', 'server/origin-shared-agents.ts', 'server/origin-shared.test.ts'].map(file => {
  const disk = fs.readFileSync(path.join(root, file));
  const committed = git('show', `HEAD:${file}`);
  if (disk.toString().replace(/\r\n/g, '\n') !== committed.toString().replace(/\r\n/g, '\n')) throw new Error(`checkout/commit mismatch: ${file}`);
  return { file, checkoutSize: disk.length, checkoutSha256: sha256(disk), committedSize: committed.length, committedSha256: sha256(committed), normalizedLfSha256: sha256(disk.toString().replace(/\r\n/g, '\n')) };
});
const verification = {
  commit: git('rev-parse', 'HEAD').toString().trim(),
  base: 'ec28ec06e935b10a176f26586348e81831475e80',
  source,
  cleanTrackedCheckout: git('status', '--porcelain=v1').length === 0,
  checks: [
    { command: 'node node_modules/typescript/lib/tsc.js -p tsconfig.server.json', exitCode: 0, log: 'tsc-server-final.log' },
    { command: 'node node_modules/oxlint/bin/oxlint --deny-warnings server/origin-run-lease.ts server/origin-shared-agents.ts server/origin-shared.test.ts', exitCode: 0, log: 'lint-final.log' },
    { command: 'git diff --check', exitCode: 0, log: 'diff-check.log' },
  ],
  claims: { uniqueBugTotal: null, ui: false, native: false, production: false, push: false },
};
fs.writeFileSync(path.join(__dirname, 'verification.json'), JSON.stringify(verification, null, 2) + '\n');
const entries = [
  'REPORT.md', 'verification.json', 'final-source-proof.json', 'red-final-account-source-proof.json',
  'run-tests.cjs', 'run-check.cjs', 'fixture-smoke.mts', 'replay-baseline-final.cjs', 'make-manifest.cjs',
  'tsc-server-final.log', 'lint-final.log', 'diff-check.log', 'change.patch',
  ...['red-final-bound', 'red-final-account', 'green-accepted-final', 'backend', 'guards-final', 'red-begin-confirmation', 'red-lifecycle'].flatMap(label => [`${label}.log`, `${label}.json`, `${label}-command.json`, `${label}-exit.json`]),
  ...['.log', '-command.json', '-exit.json', '-steps.json', '-server.log', '-cleanup.json'].map(suffix => `fixture-final${suffix}`),
  'red.log', 'red.json', 'red-exit.json', 'green-lifecycle.log', 'green-lifecycle.json', 'green-lifecycle-exit.json',
].sort();
const files = entries.map(file => {
  const data = fs.readFileSync(path.join(__dirname, file));
  return { file, size: data.length, sha256: sha256(data) };
});
const manifest = { schemaVersion: 1, commit: verification.commit, files, count: files.length, totalBytes: files.reduce((total, item) => total + item.size, 0) };
const bytes = Buffer.from(JSON.stringify(manifest, null, 2) + '\n');
fs.writeFileSync(path.join(__dirname, 'manifest.json'), bytes);
for (const entry of files) {
  const data = fs.readFileSync(path.join(__dirname, entry.file));
  if (data.length !== entry.size || sha256(data) !== entry.sha256) throw new Error(`evidence mismatch: ${entry.file}`);
}
console.log(JSON.stringify({ commit: verification.commit, entries: files.length, totalBytes: manifest.totalBytes, manifestSha256: sha256(bytes), reportSha256: sha256(fs.readFileSync(path.join(__dirname, 'REPORT.md'))) }, null, 2));
