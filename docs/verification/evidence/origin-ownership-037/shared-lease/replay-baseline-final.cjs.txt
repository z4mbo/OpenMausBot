const fs = require('node:fs');
const path = require('node:path');
const cp = require('node:child_process');
const crypto = require('node:crypto');
const root = path.resolve(__dirname, '../..');
const base = 'ec28ec06e935b10a176f26586348e81831475e80';
const label = process.argv[2] ?? 'red-final-bound';
const files = ['server/origin-run-lease.ts', 'server/origin-shared-agents.ts'];
const digest = b => crypto.createHash('sha256').update(b).digest('hex');
const snapshots = files.map(file => ({ file, current: fs.readFileSync(path.join(root, file)), baseline: cp.execFileSync('git', ['show', `${base}:${file}`], { cwd: root, maxBuffer: 10_000_000 }) }));
const testsBefore = fs.readFileSync(path.join(root, 'server/origin-shared.test.ts'));
let result;
try {
  for (const item of snapshots) fs.writeFileSync(path.join(root, item.file), item.baseline);
  const selection = process.argv[3] ?? 'stops after the last|bounds a nonsettling heartbeat|uses elapsed monotonic|does not admit a begin|confirms an idempotent begin|does not renew an existing|expires the local turn|confirms a queued promotion|never executes a queued|keeps a rejected queued|does not reuse';
  result = cp.spawnSync(process.execPath, [path.join(__dirname, 'run-tests.cjs'), label, 'server/origin-shared.test.ts', '-t', selection], { cwd: root, stdio: 'inherit', windowsHide: true });
} finally {
  for (const item of snapshots) fs.writeFileSync(path.join(root, item.file), item.current);
  const restoredExact = snapshots.every(item => fs.readFileSync(path.join(root, item.file)).equals(item.current));
  const testBytesUnchanged = fs.readFileSync(path.join(root, 'server/origin-shared.test.ts')).equals(testsBefore);
  fs.writeFileSync(path.join(__dirname, `${label}-source-proof.json`), JSON.stringify({ base, files: snapshots.map(item => ({ file: item.file, baselineSha256: digest(item.baseline), fixedSha256: digest(item.current) })), testsSha256: digest(testsBefore), restoredExact, testBytesUnchanged, redExit: result?.status, date: new Date().toISOString() }, null, 2) + '\n');
  if (!restoredExact || !testBytesUnchanged) throw new Error('source restoration or test byte equality failed');
}
if (result?.status !== 1) throw new Error(`expected genuine failing baseline; got ${result?.status}`);
