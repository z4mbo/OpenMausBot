const fs = require('node:fs');
const path = require('node:path');
const cp = require('node:child_process');
const root = path.resolve(__dirname, '../..');
const kind = process.argv[2];
const label = process.argv[3] ?? kind;
const ownedHome = path.join(__dirname, `${label}-home`);
const ownedTemp = path.join(__dirname, `${label}-temp`);
fs.mkdirSync(ownedHome, { recursive: true });
fs.mkdirSync(ownedTemp, { recursive: true });
const env = { PATH: `${path.dirname(process.execPath)};C:\\Windows\\System32;C:\\Program Files\\Git\\cmd`, SystemRoot: 'C:\\Windows', WINDIR: 'C:\\Windows', TEMP: ownedTemp, TMP: ownedTemp, HOME: ownedHome, USERPROFILE: ownedHome, APPDATA: path.join(ownedHome, 'AppData/Roaming'), LOCALAPPDATA: path.join(ownedHome, 'AppData/Local') };
const jobs = {
  backend: { cwd: path.join(root, 'backend'), args: [path.join(root, 'node_modules/vitest/vitest.mjs'), 'run', '--config', 'vitest.config.ts', 'convex/sharedAgents.test.ts', '-t', 'a runner that stops heartbeating is interrupted and its streaming rows marked', '--reporter=default', '--reporter=json', `--outputFile=${path.join(__dirname, `${label}.json`)}`] },
  guards: { cwd: root, args: [path.join(root, 'node_modules/vitest/vitest.mjs'), 'run', 'scripts/testing/file-size-ratchet.test.ts', 'scripts/testing/comment-history.test.ts', '--reporter=default', '--reporter=json', `--outputFile=${path.join(__dirname, `${label}.json`)}`] },
  fixture: { cwd: root, args: ['--experimental-strip-types', path.join(__dirname, 'fixture-smoke.mts'), label] },
};
const job = jobs[kind];
if (!job) throw new Error('unknown check');
fs.writeFileSync(path.join(__dirname, `${label}-command.json`), JSON.stringify({ node: process.execPath, ...job, env, startedAt: new Date().toISOString() }, null, 2) + '\n');
const log = fs.openSync(path.join(__dirname, `${label}.log`), 'wx');
const child = cp.spawn(process.execPath, job.args, { cwd: job.cwd, env, stdio: ['ignore', log, log], windowsHide: true });
console.log(`${kind} started: ${child.pid}`);
child.on('exit', (code, signal) => {
  fs.closeSync(log);
  fs.writeFileSync(path.join(__dirname, `${label}-exit.json`), JSON.stringify({ code, signal, finishedAt: new Date().toISOString() }, null, 2) + '\n');
  console.log(`${kind} exit ${code}`);
  process.exitCode = code ?? 1;
});
