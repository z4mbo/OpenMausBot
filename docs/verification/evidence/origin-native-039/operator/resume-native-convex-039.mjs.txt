let child, stopping = false;
async function start() {
  const output = [];
  child = spawn(record.binary, ['--interface', '127.0.0.1', '--port', String(cloudPort), '--site-proxy-port', String(sitePort), '--convex-origin', url, '--convex-site', siteUrl, '--instance-name', instance, '--instance-secret', instanceSecret, '--disable-beacon', '--local-storage', resolve(folder, 'data/storage'), resolve(folder, 'data/backend.sqlite3')], { cwd: folder, env, windowsHide: true, stdio: ['ignore', 'pipe', 'pipe'] });
  const launch = { pid: child.pid, parentPid: process.pid, startedAt: new Date().toISOString(), mode: 'same synthetic SQLite and storage, no code push' };
  record.launches.push(launch); save();
  for (const stream of [child.stdout, child.stderr]) stream.on('data', bytes => { if (output.reduce((n, item) => n + item.length, 0) < 10 * 1024 * 1024) output.push(Buffer.from(bytes)); });
  child.once('exit', (code, signal) => { Object.assign(launch, { exitCode: code, signal, closedAt: new Date().toISOString() }); writeFileSync(resolve(folder, `logs/native-${record.launches.indexOf(launch) + 1}.log`), redact(Buffer.concat(output).toString())); save(); });
  let ready = false;
  for (let i = 0; i < 100; i++) { try { const response = await fetch(`${url}/instance_name`, { signal: AbortSignal.timeout(500) }); if (response.ok && (await response.text()).replaceAll('"', '').trim() === instance) { ready = true; break; } } catch {} await new Promise(done => setTimeout(done, 200)); }
  assert.ok(ready, 'Owned backend recovery did not become ready');
  launch.readyAt = new Date().toISOString(); record.restartMode = 'Windows exact-child termination followed by same-SQLite recovery; not a graceful shutdown claim'; save();
  console.log(JSON.stringify({ restarted: true, url, siteUrl, pid: child.pid, source: record.source, codeRepushed: false }));
}
async function stop(exitCode = 0) { if (stopping) return; stopping = true; if (child && child.exitCode === null) { const done = new Promise(resolveExit => child.once('exit', resolveExit)); child.kill(); await done; } record.stoppedAt = new Date().toISOString(); save(); console.log('Owned synthetic backend stopped; data retained.'); process.exit(exitCode); }
process.on('SIGINT', stop); process.on('SIGTERM', stop);

catch (error) { record.failure = redact(String(error)); save(); console.error(record.failure); await stop(1); }
