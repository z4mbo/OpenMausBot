import assert from 'node:assert/strict';
import { execFileSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import { existsSync, readFileSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { normalizePath } from 'vite';
import { launchVerificationServer } from '../../scripts/control-omb.ts';
import { mountPreview, parkUntilSignal, REPO_ROOT } from '../../scripts/testing/preview-fixture.ts';

assert.deepEqual(process.argv.slice(2), ['--backend', 'http://127.0.0.1:51527']);
const backend = process.argv[3], folder = import.meta.dirname;
assert(!existsSync(join(folder, 'ready.json')), 'Reconcile the existing fixture before restarting');
for (const name of ['.env', '.env.local', '.env.development', '.env.development.local']) assert(!existsSync(join(REPO_ROOT, name)), 'Unexpected Vite env file');
process.env.CI = '1';
for (const name of Object.keys(process.env)) if (name.startsWith('VITE_') || name.startsWith('NEXT_PUBLIC_')) delete process.env[name];
process.env.VITE_ORIGIN_CONVEX_URL = backend;
const sourceFiles = ['src/components/OriginGate.tsx', 'src/components/OriginConnect.tsx', 'src/components/origin-members/WorkspaceMembers.tsx', 'src/lib/origin-cloud/client.ts', 'src/lib/origin-cloud/hooks.ts', 'src/lib/origin-members.ts'];
const sha = bytes => createHash('sha256').update(bytes).digest('hex');
const bindings = sourceFiles.map(file => {
  const frozen = execFileSync('git', ['show', `ab0cd72b4c2f004c1015adca034472241569c04b:${file}`], { cwd: REPO_ROOT });
  const actual = readFileSync(join(REPO_ROOT, file));
  assert.equal(actual.toString().replaceAll('\r\n', '\n'), frozen.toString().replaceAll('\r\n', '\n'));
  return { path: file, bytes: actual.length, sha256: sha(actual), frozenBlobSha256: sha(frozen) };
});
const owned = [];
try {
  for (const label of ['A', 'B']) {
    const fixtureEnvironment = { CI: '1', FAKE_CLAUDE_MODE: 'happy', OMB_TEST_PORTS: '64160-64199' };
    for (const key of ['SYSTEMROOT', 'WINDIR', 'COMSPEC', 'PATHEXT', 'LANG', 'LC_ALL', 'TZ']) if (process.env[key]) fixtureEnvironment[key] = process.env[key];
    const fixture = await launchVerificationServer(fixtureEnvironment);
    const record = { label, fixture, preview: null };
    owned.push(record);
    const csp = `default-src 'self'; script-src 'self' 'unsafe-inline'; style-src 'self' 'unsafe-inline'; img-src 'self' data: blob:; font-src 'self' data:; connect-src 'self' ${backend} ${backend.replace('http:', 'ws:')}; form-action 'self'; base-uri 'none'; frame-ancestors 'none'`;
    const calls = [];
    const entry = normalizePath(join(folder, 'entry.tsx'));
    record.preview = await mountPreview(fixture, {
      entry: '/__origin-local-entry.tsx', route: '/__origin-local-two-client.html', title: `Origin local client ${label}`, hmr: false,
      cacheDir: join(fixture.info.dataDir, 'local-client-vite-cache'),
      plugins: [{ name: 'owned-local-entry', enforce: 'pre',
        resolveId(id) { if (id === '/__origin-local-entry.tsx') return '\0origin-local-entry.tsx'; },
        load(id) { if (id === '\0origin-local-entry.tsx') return readFileSync(entry, 'utf8'); },
        configureServer(server) {
          server.middlewares.use((req, res, next) => {
            res.setHeader('Content-Security-Policy', csp);
            const pathname = (req.url ?? '').split('?')[0];
            if (/^\/(?:oauth|join|feedback)\/|^\/\.well-known\/oauth-/.test(pathname)) { res.statusCode = 404; res.end('Unavailable in local fixture'); return; }
            if (/^\/api\/origin\/(?:cloud-session|invites|invite-address)$/.test(pathname)) res.once('finish', () => {
              calls.push({ method: req.method, path: pathname, status: res.statusCode, at: new Date().toISOString() });
              writeFileSync(join(folder, `client-${label}-local-route-observations.json`), JSON.stringify(calls, null, 2) + '\n');
            });
            next();
          });
        },
      }],
      extraRoutes: [{ path: '/__origin-local-contract.json', method: 'GET', handler(_req, res) {
        res.setHeader('content-type', 'application/json'); res.end(JSON.stringify({ label, backend }));
      } }],
    });
  }
  const clients = owned.map(({ label, fixture, preview }) => {
    const url = new URL(preview.previewUrl); if (label === 'B') url.hostname = 'localhost';
    return { label, url: url.href, api: fixture.info, storageOrigin: url.origin };
  });
  assert.notEqual(new URL(clients[0].url).hostname, new URL(clients[1].url).hostname);
  const ready = { backend, frozenUiSource: 'ab0cd72b4c2f004c1015adca034472241569c04b', checkout: execFileSync('git', ['rev-parse', 'HEAD'], { cwd: REPO_ROOT, encoding: 'utf8' }).trim(), clients, bindings, actualConvexTransport: true, authOrMembershipMocks: false, backendMutationsFromLauncher: 0, mailKeysSupplied: false, parentPid: process.pid, readyAt: new Date().toISOString() };
  writeFileSync(join(folder, 'ready.json'), JSON.stringify(ready, null, 2) + '\n', { flag: 'wx' });
  console.log(JSON.stringify({ backend, clients: clients.map(client => ({ label: client.label, url: client.url, fixturePid: client.api.pid, fixtureUrl: client.api.url })), parentPid: process.pid }));
  await parkUntilSignal();
} finally {
  for (const client of owned.reverse()) { await client.preview?.close(); await client.fixture.close(); }
}
