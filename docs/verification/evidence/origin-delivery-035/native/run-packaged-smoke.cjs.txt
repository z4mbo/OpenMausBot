const fs = require('node:fs');
const path = require('node:path');
const cp = require('node:child_process');
const crypto = require('node:crypto');
const repo = path.resolve(__dirname, '../..');
const node = 'C:/Users/aless/Documents/Origin-bugfix/.omb-scratch/runtimes/node-v24.21.0-win-x64/node.exe';
const resources = path.join(__dirname, 'windows/resources');
const home = path.join(__dirname, 'smoke-own-home');
const temp = path.join(__dirname, 'smoke-own-temp');
for (const directory of [home,temp,path.join(home,'AppData/Roaming'),path.join(home,'AppData/Local')]) fs.mkdirSync(directory,{recursive:true});
const env = {
  PATH:'C:\\Windows\\System32',
  SystemRoot:'C:\\Windows',
  WINDIR:'C:\\Windows',
  TEMP:temp,
  TMP:temp,
  HOME:home,
  USERPROFILE:home,
  APPDATA:path.join(home,'AppData/Roaming'),
  LOCALAPPDATA:path.join(home,'AppData/Local'),
  XDG_CONFIG_HOME:path.join(home,'.config'),
  XDG_CACHE_HOME:path.join(home,'.cache'),
  XDG_DATA_HOME:path.join(home,'.local/share'),
  OMB_DATA_DIR:path.join(home,'.openmausbot'),
  OMB_SMOKE_DIST:path.join(resources,'server'),
};
const script = path.join(repo,'scripts/smoke-packaged-server.mjs');
const args = [script,'--browser-bundle',path.join(resources,'browser-engine')];
const command = {node,nodeVersion:cp.execFileSync(node,['--version'],{env,encoding:'utf8'}).trim(),args,cwd:repo,env,scriptSha256:crypto.createHash('sha256').update(fs.readFileSync(script)).digest('hex'),startedAt:new Date().toISOString()};
fs.writeFileSync(path.join(__dirname,'smoke-command.json'),JSON.stringify(command,null,2)+'\n');
const log = fs.openSync(path.join(__dirname,'packaged-smoke.log'),'wx');
const child = cp.spawn(node,args,{cwd:repo,env,stdio:['ignore',log,log],windowsHide:true});
command.pid = child.pid;
fs.writeFileSync(path.join(__dirname,'smoke-command.json'),JSON.stringify(command,null,2)+'\n');
console.log(`Isolated downloaded-package smoke started: pid ${child.pid}`);
child.on('error',(error)=>{fs.closeSync(log);fs.writeFileSync(path.join(__dirname,'smoke-result.json'),JSON.stringify({error:String(error),finishedAt:new Date().toISOString()},null,2)+'\n');process.exitCode=1;});
child.on('exit',(code,signal)=>{fs.closeSync(log);const result={code,signal,pid:child.pid,finishedAt:new Date().toISOString(),remainingTempEntries:fs.readdirSync(temp)};fs.writeFileSync(path.join(__dirname,'smoke-result.json'),JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result,null,2));process.exitCode=code??1;});
