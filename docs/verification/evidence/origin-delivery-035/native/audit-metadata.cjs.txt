const assert = require('node:assert/strict');
const fs = require('node:fs');
const path = require('node:path');
const crypto = require('node:crypto');
const cp = require('node:child_process');
const repo = path.resolve(__dirname, '../..');
const asar = require(path.join(repo, 'node_modules/.pnpm/@electron+asar@3.4.1/node_modules/@electron/asar/lib/asar.js'));
const plist = require(path.join(repo, 'node_modules/.pnpm/plist@3.1.0/node_modules/plist'));
const yaml = require(path.join(repo, 'node_modules/.pnpm/yaml@2.9.0/node_modules/yaml'));
const frozen = '1483a7d76fb092b5ef6860afc5ffb520fc4e7707';
const sha = (bytes) => crypto.createHash('sha256').update(bytes).digest('hex');
const normalized = (bytes) => Buffer.from(bytes.toString('utf8').replace(/\r\n/g, '\n'));
const source = (filename) => cp.execFileSync('git', ['show', `${frozen}:${filename}`], { cwd: repo, maxBuffer: 20 * 1024 * 1024 });
const args = JSON.parse(fs.readFileSync(process.argv[2], 'utf8'));
const results = [];
for (const target of args.targets) {
  const resources = target.platform === 'mac' ? path.join(target.extractedTo, 'Origin.app/Contents/Resources') : path.join(target.extractedTo, target.resourceRelative);
  const appAsar = path.join(resources, 'app.asar');
  const entries = asar.listPackage(appAsar).map((entry) => entry.replace(/^[/\\]+/, '').replace(/\\/g, '/'));
  assert(entries.includes('shared/product-brand.mjs'), `${target.label} omitted brand module`);
  const brand = asar.extractFile(appAsar, 'shared/product-brand.mjs');
  assert.deepEqual(normalized(brand), normalized(source('shared/product-brand.mjs')), `${target.label} brand module differs`);
  const pkg = JSON.parse(asar.extractFile(appAsar, 'package.json').toString('utf8'));
  assert.equal(pkg.name, 'origin');
  assert.equal(pkg.version, '0.2.13');
  assert.equal(pkg.author.name, 'Origin');
  const updater = yaml.parse(fs.readFileSync(path.join(resources, 'app-update.yml'), 'utf8'));
  assert.equal(updater.provider, 'generic');
  assert.equal(updater.url, 'https://origin.imbored.fun/downloads/');
  const compared = [];
  for (const entry of entries.filter((entry) => /^electron\/.+\.(mjs|cjs)$/.test(entry))) {
    const expected = source(entry);
    const actual = asar.extractFile(appAsar, path.normalize(entry));
    assert.deepEqual(normalized(actual), normalized(expected), `${target.label}: ${entry} differs from frozen source`);
    compared.push({path:entry, bytes:actual.length, packagedSha256:sha(actual), sourceSha256:sha(expected), normalizedSha256:sha(normalized(actual))});
  }
  assert.equal(compared.length,159,`${target.label}: expected frozen plain Electron file count`);
  const record = {label:target.label, platform:target.platform, resources, asarSha256:sha(fs.readFileSync(appAsar)), asarEntries:entries.length, package:{name:pkg.name,version:pkg.version,author:pkg.author,main:pkg.main}, updater, brand:{path:'shared/product-brand.mjs',bytes:brand.length,sha256:sha(brand),normalizedSha256:sha(normalized(brand))},plainElectronSources:compared};
  if (target.platform === 'mac') {
    const info = plist.parse(fs.readFileSync(path.join(target.extractedTo, 'Origin.app/Contents/Info.plist'), 'utf8'));
    assert.equal(info.CFBundleName, 'Origin');
    assert.equal(info.CFBundleDisplayName, 'Origin');
    assert.equal(info.CFBundleIdentifier, 'com.imbored.origin');
    assert.equal(info.CFBundleExecutable, 'Origin');
    assert.equal(info.CFBundleShortVersionString, '0.2.13');
    assert.equal(info.CFBundleVersion, '0.2.13');
    assert(info.CFBundleURLTypes.some((type) => type.CFBundleURLSchemes.includes('origin-app')));
    const executable = fs.readFileSync(path.join(target.extractedTo, 'Origin.app/Contents/MacOS/Origin'));
    assert.equal(executable.readUInt32LE(0), 0xfeedfacf);
    assert.equal(executable.readUInt32LE(4), target.arch === 'arm64' ? 0x0100000c : 0x01000007);
    const sentinel = fs.readFileSync(path.join(resources, 'server/openmausbot.js'));
    assert(sentinel.toString('utf8').includes('--check-install'), `${target.label} CLI fails exact Sentinel advertisement predicate`);
    record.mac = {bundleName:info.CFBundleName,displayName:info.CFBundleDisplayName,bundleIdentifier:info.CFBundleIdentifier,executable:info.CFBundleExecutable,version:info.CFBundleShortVersionString,buildVersion:info.CFBundleVersion,protocols:info.CFBundleURLTypes,sentinelPaths:{node:'Origin.app/Contents/MacOS/Origin',script:'Origin.app/Contents/Resources/server/openmausbot.js'},executableSha256:sha(executable),machOCpuType:executable.readUInt32LE(4),sentinelCliBytes:sentinel.length,sentinelCliSha256:sha(sentinel)};
  }
  results.push(record);
}
for (const feed of args.feeds) {
  const content = yaml.parse(fs.readFileSync(feed.path, 'utf8'));
  assert.equal(content.version, '0.2.13');
  assert(content.files.length > 0);
  assert(content.files.every((file) => file.url.startsWith('Origin-0.2.13-')));
  feed.metadata = {version:content.version,path:content.path,files:content.files.map((file) => ({url:file.url,size:file.size})),releaseNotesPresent:typeof content.releaseNotes === 'string' && content.releaseNotes.length > 0};
}
const report = {frozenSource:frozen,createdAt:new Date().toISOString(),normalization:'CRLF to LF only',toolVersions:{asar:'3.4.1',plist:'3.1.0',yaml:'2.9.0'},targets:results,feeds:args.feeds};
fs.writeFileSync(path.join(__dirname, 'metadata-results.json'), JSON.stringify(report,null,2)+'\n');
console.log(JSON.stringify({frozenSource:frozen,targets:results.map((record)=>({label:record.label,version:record.package.version,plainElectronFiles:record.plainElectronSources.length,brandSha256:record.brand.sha256,asarSha256:record.asarSha256})),feeds:args.feeds.map((feed)=>({label:feed.label,...feed.metadata}))},null,2));
